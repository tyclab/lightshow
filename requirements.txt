# Python dependencies of the live input (src/live_input.py, src/analysis/): it
# hears the music as it plays and tells the server the beat, the bands and the
# events the auto show answers.
#
#     pip install -r requirements.txt
#
# into the Python the server uses (Sources → Live input → Python, or the first
# python3 on PATH, or a .venv in the project folder).

numpy>=1.24

# soundcard captures what the PC itself plays (WASAPI loopback on Windows, the
# output's monitor on Linux) and any input; sounddevice is its fallback for a
# line-in. Both load their system audio library when first used, not when
# installed, so a machine without one only loses the live input.
soundcard>=0.4.6
sounddevice>=0.5

# Reading an audio file instead of a device (live_input.py --file), which is
# what the tests do.
librosa>=0.10
soundfile>=0.12
